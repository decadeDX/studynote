function formatWeeks(weeks) {

    if (!weeks || weeks.length === 0) {
        return "";
    }

    const sorted = [...weeks].sort((a, b) => a - b);

    if (sorted.length === 1) {
        return `${sorted[0]}`;
    }

    let continuous = true;

    for (let i = 1; i < sorted.length; i++) {

        if (sorted[i] !== sorted[i - 1] + 1) {
            continuous = false;
            break;
        }
    }

    if (continuous) {
        return `${sorted[0]}~${sorted[sorted.length - 1]}`;
    }

    return sorted.join(",");
}

/* =========================================================

   ========================================================= */

function render() {

    root.innerHTML = "";


    /* -----------------------------------------------------
       当前周日期
       ----------------------------------------------------- */

    const weekStart =
        START_DATE.plus({
            days: (currentWeek - 1) * 7
        });

    const weekEnd =
        weekStart.plus({ days: 6 });


    /* =====================================================
       顶部
       ===================================================== */

    const toolbar =
        create("div", "tt-toolbar");


    /* 左侧标题 */

    const titleBox =
        create("div", "tt-title-box");

    const title =
        create("div", "tt-title");

    title.textContent =
        "📅 2026-2027 秋季学期";


    const rangeText =
        create("div", "tt-range");

    rangeText.textContent =
        `${weekStart.toFormat("yyyy年M月d日")} — ${weekEnd.toFormat("yyyy年M月d日")}`;


    titleBox.appendChild(title);
    titleBox.appendChild(rangeText);


    /* 右侧按钮 */

    const controls =
        create("div", "tt-controls");


    const prev =
        create("button", "tt-button tt-arrow");

    prev.textContent = "‹";


    const weekDisplay =
        create("div", "tt-week-display");

    weekDisplay.innerHTML =
        `第 ${currentWeek} 周
        <span>
            (${weekStart.toFormat("M.d")} - ${weekEnd.toFormat("M.d")})
        </span>`;


    const next =
        create("button", "tt-button tt-arrow");

    next.textContent = "›";


    const current =
        create("button", "tt-button tt-current");

    current.textContent = "本周";


    prev.onclick = () => {

        if (currentWeek > 1) {

            currentWeek--;

            sessionStorage.setItem(
                STORAGE_KEY,
                String(currentWeek)
            );

            render();
        }
    };


    next.onclick = () => {

        if (currentWeek < TOTAL_WEEKS) {

            currentWeek++;

            sessionStorage.setItem(
                STORAGE_KEY,
                String(currentWeek)
            );

            render();
        }
    };


    current.onclick = () => {

        currentWeek = autoWeek;

        sessionStorage.setItem(
            STORAGE_KEY,
            String(currentWeek)
        );

        render();
    };


    controls.appendChild(prev);
    controls.appendChild(weekDisplay);
    controls.appendChild(next);
    controls.appendChild(current);


    toolbar.appendChild(titleBox);
    toolbar.appendChild(controls);

    root.appendChild(toolbar);


    /* =====================================================
       滚动容器
       ===================================================== */

    const scroll =
        create("div", "tt-scroll");


    /* =====================================================
       Table
       ===================================================== */

    const table =
        create("table", "tt-table");


    /* =====================================================
       THEAD
       ===================================================== */

    const thead =
        document.createElement("thead");

    const headerRow =
        document.createElement("tr");


    /* 左上角 */

    const corner =
        document.createElement("th");

    corner.className = "tt-corner";

    headerRow.appendChild(corner);


    const weekdays = [
        "星期一",
        "星期二",
        "星期三",
        "星期四",
        "星期五",
        "星期六",
        "星期日"
    ];


    for (let day = 1; day <= 7; day++) {

        const date =
            weekStart.plus({
                days: day - 1
            });

        const th =
            document.createElement("th");

        th.className = "tt-day-head";


        if (day === 6) {
            th.classList.add("tt-saturday");
        }

        if (day === 7) {
            th.classList.add("tt-sunday");
        }

        if (date.hasSame(today, "day")) {
            th.classList.add("tt-today");
        }


        const weekName =
            create("div", "tt-week-name");

        weekName.textContent =
            weekdays[day - 1];


        const dateName =
            create("div", "tt-date-name");

        dateName.textContent =
            date.toFormat("M月d日");


        th.appendChild(weekName);
        th.appendChild(dateName);

        headerRow.appendChild(th);
    }


    thead.appendChild(headerRow);

    table.appendChild(thead);


    /* =====================================================
       当前周课程
       ===================================================== */

    const weekCourses =
        courses.filter(course =>
            course.weeks.includes(currentWeek)
        );


    /*
       occupied[weekday][period]

       start = 课程起点
       skip  = 被 rowspan 覆盖
    */

    const occupied = {};

    for (let day = 1; day <= 7; day++) {

        occupied[day] = {};

        for (let period = 1; period <= PERIODS; period++) {

            occupied[day][period] = null;
        }
    }


    for (const course of weekCourses) {

        occupied[course.weekday][course.start] = {
            kind: "start",
            course: course
        };


        for (
            let period = course.start + 1;
            period <= course.end;
            period++
        ) {

            occupied[course.weekday][period] = {
                kind: "skip",
                course: course
            };
        }
    }


    /* =====================================================
       TBODY
       ===================================================== */

    const tbody =
        document.createElement("tbody");


    for (
        let period = 1;
        period <= PERIODS;
        period++
    ) {

        const tr =
            document.createElement("tr");


        /* -------------------------------------------------
           节次
           ------------------------------------------------- */

        const periodCell =
            document.createElement("td");

        periodCell.className =
            "tt-period-cell";


        if (period <= 4) {

            periodCell.classList.add(
                "tt-morning"
            );

        } else if (period <= 10) {

            periodCell.classList.add(
                "tt-afternoon"
            );

        } else {

            periodCell.classList.add(
                "tt-evening"
            );
        }


        // 节次数字不写入 HTML，由 CSS counter 自动生成
        periodCell.textContent = "";

        tr.appendChild(periodCell);


        /* -------------------------------------------------
           周一～周日
           ------------------------------------------------- */

        for (
            let day = 1;
            day <= 7;
            day++
        ) {

            const state =
                occupied[day][period];


            /* 已被上方 rowspan 覆盖 */

            if (
                state &&
                state.kind === "skip"
            ) {

                continue;
            }


            const td =
                document.createElement("td");

            td.className =
                "tt-course-cell";


            if (day >= 6) {

                td.classList.add(
                    "tt-weekend-cell"
                );
            }


            /* ---------------------------------------------
               课程
               --------------------------------------------- */

            if (
                state &&
                state.kind === "start"
            ) {

                const course =
                    state.course;


                const span =
                    course.end -
                    course.start +
                    1;


                td.rowSpan = span;

                td.classList.add(
                    "tt-has-course"
                );


                const card =
                    create(
                        "div",
                        `tt-course-card tt-${course.type}`
                    );


                /* 课程名 */

                const courseTitle =
                    create(
                        "div",
                        "tt-course-title"
                    );


                const badge =
                    create(
                        "span",
                        "tt-badge"
                    );

                badge.textContent = "硕";


                const courseName =
                    create(
                        "span",
                        "tt-course-name"
                    );

                courseName.textContent =
                    course.name;


                courseTitle.appendChild(badge);
                courseTitle.appendChild(courseName);

                card.appendChild(courseTitle);


                /* 代码 */

                if (course.code) {

                    const code =
                        create(
                            "div",
                            "tt-course-code"
                        );

                    code.textContent =
                        course.code;

                    card.appendChild(code);
                }


                /* 周次节次 */

                const schedule =
                    create(
                        "div",
                        "tt-course-info"
                    );

                schedule.textContent =
                    `(${formatWeeks(course.weeks)}周) (${course.start}-${course.end}节)`;

                card.appendChild(schedule);


                /* 教室教师 */

                const locationParts = [];

                if (course.room) {
                    locationParts.push(
                        course.room
                    );
                }

                if (course.teacher) {
                    locationParts.push(
                        course.teacher
                    );
                }


                if (locationParts.length) {

                    const location =
                        create(
                            "div",
                            "tt-course-info"
                        );

                    location.textContent =
                        locationParts.join("　");

                    card.appendChild(location);
                }


                /* 班级 */

                if (course.className) {

                    const className =
                        create(
                            "div",
                            "tt-course-class"
                        );

                    className.textContent =
                        course.className;

                    card.appendChild(className);
                }


                td.appendChild(card);
            }


            tr.appendChild(td);
        }


        tbody.appendChild(tr);
    }


    table.appendChild(tbody);

    scroll.appendChild(table);

    root.appendChild(scroll);


    /* =====================================================
       Footer
       ===================================================== */

    const footer =
        create("div", "tt-footer");


    const status =
        create("div", "tt-status");

    status.innerHTML =
        `ℹ️ 当前显示：
         <strong>第 ${currentWeek} 周</strong>
         ·
         ${weekStart.toFormat("M月d日")}
         -
         ${weekEnd.toFormat("M月d日")}`;


    const legend =
        create("div", "tt-legend");


    const legendItems = [
        ["general", "通识课"],
        ["major", "专业课"],
        ["experiment", "实验/实践"],
        ["language", "语言类"]
    ];


    for (
        const [type, label]
        of legendItems
    ) {

        const item =
            create("span", "tt-legend-item");

        const dot =
            create(
                "i",
                `tt-legend-dot tt-dot-${type}`
            );

        const text =
            document.createElement("span");

        text.textContent = label;


        item.appendChild(dot);
        item.appendChild(text);

        legend.appendChild(item);
    }


    footer.appendChild(status);
    footer.appendChild(legend);

    root.appendChild(footer);
}


/* =========================================================
   首次运行
   ========================================================= */
